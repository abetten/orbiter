		-povray \
		-round 0 -nb_frames_default 30 \
		-output_mask dode_%d_%03d.pov \

			-group_of_things_as_interval 0 20 \

			-group_of_things_as_interval 0 12\

		-scene_objects_end \
		-povray_end
