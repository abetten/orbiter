Hamming_weight_enumerator:
	$(ORBITER) -v 2 \
		-define F -finite_field -q 2 -end \
		-define v -vector -field F -format 4 \
			-dense $(HAMMING_CODE_GENERATOR) \
		-end \
		-define C -code -field F \
			-generator_matrix v \
		-end \
		-with C -do \
		-coding_theoretic_activity \
			-weight_enumerator \
		-end
