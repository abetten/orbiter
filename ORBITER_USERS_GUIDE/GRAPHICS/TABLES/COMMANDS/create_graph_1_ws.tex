Command & Arguments & Purpose \\
\hline
\hline
\verb'-adjacency_matrix'  &  vector   &  Create graph from the adjacency matrix. A vector object must be given. \\
\hline
\verb'-load'  &  filename   &  Read a graph from file in Orbiter format. \\
\hline
\verb'-load_csv_no_border'  &  filename   &  Read a graph from a csv file. The first row and the first column are part of the adjacency matrix. \\
\hline
\verb'-load_adjacency_matrix_from_csv_' \verb'and_select_value'  &  filename $v$  &  Read a graph from a csv file. The first row and the first column are ignored. Entries equal to $v$ define adjacencies. All other values are ignored. \\
\hline
\verb'-load_dimacs'  &  filename   & Read a graph from file in dimacs format. \\
\hline
\verb'-load_Brouwer'  &  filename   & Read a graph from file in Brouwer format. \\
\hline
\verb'-edge_list'  &  $n$ list-of-edges   & Create a graph on $n$ vertices from a list of edges as ranked pairs. \\
\hline
\verb'-edges_as_pairs'  &  $n$ edges-as-pairs   & Create a graph on $n$ vertices from a list of edges as pairs. \\
\hline
\verb'-cycle'  &  $n$    & Cycle graph on $n$ vertices. \\
\hline
\verb'-inversion_graph'  &  text    & Read inversion graph. \\
\hline
\verb'-Hamming'  &  $n$ $q$   & Hamming graph $H(n,q)$ \\
\hline
\verb'-Johnson'  &  $n$ $k$ $s$   & Johnson graph \\
\hline
\verb'-Paley'  &  $F$   & Paley graph over the field $F=\bbF_q$. \\
\hline
\verb'-Sarnak'  &  $p$ $q$   & Lubotzky-Phillips-Sarnak graph~\cite{Lubotzky1988} \\
\hline
\verb'-Schlaefli'  &  $F$   & Schlaefli graph \\
\hline
\verb'-Shrikhande'  &     & Shrikhande graph \\
\hline
\verb'-Winnie_Li'  &  $F$ $i$   & Winnie-Li graph (see~\cite{Li1992}) over the field $F=\bbF_q$ with subfield index $i$. \\
\hline
\verb'-Grassmann'  &  $n$ $k$ $F$ $r$   & Grassmann graph over $F = \bbF_q$. \\
\hline
\verb'-coll_orthogonal'  &  $O$ $S$    & Collinearity graph of the set $S$ in the orthogonal geometry $O$. \\
\hline
\verb'-affine_polar'  &  $O$    & Affine polar graph associated to the orthogonal geometry $O$. \\
\hline
\verb'-trihedral_pair_disjointness_graph'  &      & Trihedral pair disjointness graph.\\
\hline
\verb'-non_attacking_queens_graph'  &   $n$   & Create the graph for non-attacking queens on a $n \times n$ chess board. \\
