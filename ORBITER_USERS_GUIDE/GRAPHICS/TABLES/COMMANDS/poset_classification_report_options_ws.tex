\hline
Command & Arguments & Purpose \\
\hline
\hline
\verb'-select_orbits_by_level'  &   level   &  Select the levels which should be drawn. \\
\hline
\verb'-select_orbits_by_stabilizer_order'  &  m    &  Select the orbits whose stabilizer order is equal to $m$. \\
\hline
\verb'-select_orbits_by_stabilizer_order_' \verb'multiple_of'  &   m   &  Select the orbits whose stabilizer order is a multiple of $m$. \\
\hline
\verb'-include_projective_stabilizer'  &      &  Include the projective stabilizer in the report. This option assumes that the acting group is a collineation group. The projective stabilizer is a subgroup of the collineation stabilizer. \\
\hline
\verb'-draw_poset'  &      &  Include a drawing of the poset of orbits in the report. This option requires one of the following four options to set the type of drawing. \\
\hline
\verb'-type_aux'  &      &  Poset drawing of auxiliary type. The auxiliary poset includes the flag orbits between layers. \\
\hline
\verb'-type_ordinary'  &      &  Poset drawing of ordinary type. \\
\hline
\verb'-type_tree'  &      &  Poset drawing of type spanning tree. \\
\hline
\verb'-type_detailed'  &      &  Poset drawing of type detailed. The detailed drawing includes the flag orbits between layers.  \\
\hline
\verb'-fname'  &   fname   &  Set the file name. \\
\hline
\verb'-draw_options'  &   label   &  Set draw options. label must refer to an object of type \verb'draw_options', see Table~\ref{tab:draw:options}. \\
\hline
\end{tcolorbox}}
