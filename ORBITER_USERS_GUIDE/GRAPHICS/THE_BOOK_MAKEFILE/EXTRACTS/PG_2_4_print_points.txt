PG_2_4_print_points:
	$(ORBITER) -v 6 \
		-define Pts -vector \
			-loop 0 21 1 \
		-end \
		-define F -finite_field -q 4 -end \
		-define P -projective_space -n 2 -field F -v 0 -end \
		-with P -do \
		-projective_space_activity \
			-print_points Pts \
		-end
	$(PDFLATEX) PG_2_4_points.tex
	$(OPEN) PG_2_4_points.pdf
