		-define F -finite_field -q 5 -end \
		-define O -orthogonal_space 0 5 F -end \

			-space O -catalogue 1 \
		-end \
		-with BLT -do -blt_set_activity \
			-report \
		-end
	$(PDFLATEX) BLT_catalogue_q5_iso1.tex
	$(OPEN) BLT_catalogue_q5_iso1.pdf
