quartic_curves_19_report: 
	$(ORBITER) -v 3 \
		-define F -finite_field -q 19 -end \
		-define P -projective_space -n 2 -field F -v 0 -end \
		-loop L 0 $(NB_QUARTIC_CURVES_Q19) 1 \
		-define C -quartic_curve -space P -catalogue %L -end \
		-with C -do \
			-quartic_curve_activity \
				-report \
			-end \
		-end_loop L
