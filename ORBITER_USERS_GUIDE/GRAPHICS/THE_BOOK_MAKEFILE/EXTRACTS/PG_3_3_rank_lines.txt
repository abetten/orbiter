		-dense "1,0,2,2,0,1,1,2, 1,0,2,0,0,1,1,2, 1,0,2,2,0,1,2,1" \
		-end \

		-with P -do \
		-projective_space_activity \
			-rank_lines_in_PG v2 \
		-end
