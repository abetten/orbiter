symbolic_CRC32:
	$(ORBITER) -v 3 \
		-define F -finite_field -q 2 -end \
		-define M -symbolic_object \
			-field F \
			-text $(CRC32_ETHERNET_POLY) \
		-end \
		-define coeffs -symbolic_object \
			-field F \
			-collect M X \
		-end \
		-with coeffs -do -symbolic_object_activity \
			-print \
		-end
