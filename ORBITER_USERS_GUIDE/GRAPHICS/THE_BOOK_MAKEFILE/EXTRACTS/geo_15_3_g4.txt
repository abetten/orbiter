		-define Test_lines -set -loop 4 16 1 -end \

			-fuse 1 -fname_GEO 15_3_g4 \

			-nodes_empty \
			-scale 0.5 -line_width 0.3 -end \

			-file 15_3_g4_tree.txt \
			-draw_options Do \
		-end
