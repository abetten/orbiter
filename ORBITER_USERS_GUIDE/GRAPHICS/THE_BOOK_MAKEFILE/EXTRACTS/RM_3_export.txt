		-define F -finite_field -q 2 -end \
		-define C -code -field F \
			-Reed_Muller 3 \
		-end \

			-export_magma RM_3.magma \
		-end \
		-with C -and F -do -coding_theoretic_activity \
			-export_codewords RM_3_codewords.csv \
		-end
