			-fname_GEO 10_3 \
			-output_to_inc_file \

		-end

		-define Doinc -draw_incidence_structure_options \
			-width 60 -width_10 6 -end \

			-label geo_10_3 geo\_10\_3 \

		-combinatorial_object_activity \
			-canonical_form \

			-end \
		-end \
		-with C -do \
		-combinatorial_object_activity \
			-post_processing \
		-end \

				-export_flag_orbits \
				-show_incidence_matrices \
				-export_group_GAP \
