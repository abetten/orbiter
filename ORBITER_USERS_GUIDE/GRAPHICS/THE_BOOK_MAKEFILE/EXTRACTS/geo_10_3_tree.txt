geo_10_3_tree:
	$(ORBITER) -v 20 \
		-define Test_lines -set -loop 0 11 1 -end \
		-define GEO -geometry_builder \
			-V 10 -B 10 -TDO 3 -fuse 1 \
			-fname_GEO 10_3 \
			-output_to_inc_file \
			-search_tree \
			-test Test_lines \
		-end
	$(ORBITER) -v 20 \
		-define Do -draw_options -embedded -radius 40 \
			-paperheight 220 \
			-paperwidth 330 \
			-xin 10000 -yin 10000 \
			-xout 1000000 -yout 500000 \
			-scale 2 -line_width 0.3 \
			-nodes_empty \
		-end \
		-tree_draw \
			-file 10_3_tree.txt \
			-draw_options Do \
		-end
	$(PDFLATEX) 10_3_tree_draw.tex
	$(OPEN) 10_3_tree_draw.pdf
